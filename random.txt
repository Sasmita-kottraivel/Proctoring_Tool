#define IR_SENSOR_1 4   // bottle presence
#define IR_SENSOR_2 5   // drop detection

void setup() {
  Serial.begin(115200);
  pinMode(IR_SENSOR_1, INPUT);
  pinMode(IR_SENSOR_2, INPUT);
  Serial.println("IR sensors ready. Waving hand in front should change readings.");
}

void loop() {
  int val1 = digitalRead(IR_SENSOR_1);
  int val2 = digitalRead(IR_SENSOR_2);

  Serial.print("IR1 (bottle): ");
  Serial.print(val1 == LOW ? "DETECTED" : "clear");
  Serial.print("   |   IR2 (drop): ");
  Serial.println(val2 == LOW ? "DETECTED" : "clear");

  delay(300);
}
